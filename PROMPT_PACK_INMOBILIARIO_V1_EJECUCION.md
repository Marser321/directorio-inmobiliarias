import { NextResponse } from 'next/server';
import { insforge } from 'insforge';

export async function POST(request: Request) {
    const { email, password } = await request.json();

    try {
        const { accessToken } = await insforge.auth.signInWithPassword({ email, password });

        // Guardar el accessToken en una cookie httpOnly
        const response = NextResponse.json({ message: 'Login successful' });
        response.cookies.set('accessToken', accessToken, {
            httpOnly: true,
            secure: process.env.NODE_ENV === 'production', // Solo en producción
            path: '/',
            maxAge: 60 * 60 * 24, // 1 día
        });

        return response;
    } catch (error) {
        return NextResponse.json({ error: error.message }, { status: 401 });
    }
}

import { NextResponse } from 'next/server';
import { insforge } from 'insforge';

export async function GET(request: Request) {
    const accessToken = request.cookies.get('accessToken')?.value;

    if (!accessToken) {
        return NextResponse.json({ user: null }, { status: 401 });
    }

    try {
        const user = await insforge.auth.getCurrentUser(accessToken);
        return NextResponse.json({ user });
    } catch (error) {
        return NextResponse.json({ error: error.message }, { status: 401 });
    }
}

import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
    const accessToken = request.cookies.get('accessToken')?.value;

    // Proteger la ruta /dashboard
    if (request.nextUrl.pathname.startsWith('/dashboard') && !accessToken) {
        return NextResponse.redirect(new URL('/login', request.url));
    }

    return NextResponse.next();
}

export const config = {
    matcher: ['/dashboard/:path*'],
};

Aquí tienes un flujo MVP en TypeScript para InsForge que cumple con los requisitos que mencionaste. Este código incluye la inserción de una propiedad, la subida de imágenes a un bucket y el almacenamiento de la información de las fotos en la base de datos, manejando errores y validaciones.

```typescript
import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Property } from './entities/property.entity';
import { PropertyPhoto } from './entities/property-photo.entity';
import { S3 } from 'aws-sdk';
import { v4 as uuidv4 } from 'uuid';

@Injectable()
export class PropertyService {
  private s3: S3;

  constructor(
    @InjectRepository(Property)
    private propertyRepository: Repository<Property>,
    @InjectRepository(PropertyPhoto)
    private propertyPhotoRepository: Repository<PropertyPhoto>,
  ) {
    this.s3 = new S3();
  }

  async createPropertyWithPhotos(propertyData: any, photos: Express.Multer.File[]) {
    // Validación de fotos
    if (photos.length > 10) {
      throw new Error('Se permiten un máximo de 10 fotos.');
    }

    const property = this.propertyRepository.create(propertyData);
    try {
      // Insertar propiedad
      const savedProperty = await this.propertyRepository.save(property);
      const propertyId = savedProperty.id;

      const photoPromises = photos.map(async (photo, index) => {
        if (photo.size > 5 * 1024 * 1024) {
          throw new Error(`La foto ${photo.originalname} excede el tamaño máximo de 5MB.`);
        }

        const timestamp = Date.now();
        const filename = `${timestamp}-${photo.originalname}`;
        const key = `properties/${propertyId}/${filename}`;

        // Subir imagen a S3
        const uploadResult = await this.s3.upload({
          Bucket: 'property-photos',
          Key: key,
          Body: photo.buffer,
          ContentType: photo.mimetype,
        }).promise();

        // Guardar información de la foto
        const propertyPhoto = this.propertyPhotoRepository.create({
          property_id: propertyId,
          photo_url: uploadResult.Location,
          photo_key: uploadResult.Key,
          sort_order: index,
        });

        return this.propertyPhotoRepository.save(propertyPhoto);
      });

      await Promise.all(photoPromises);
      return savedProperty;
    } catch (error) {
      // Manejo de errores y rollback
      await this.propertyRepository.delete(property.id); // Borrar propiedad si falla
      throw new Error(`Error al crear la propiedad: ${error.message}`);
    }
  }
}
```

### Explicación del código:

1. **Validación de fotos**: Se verifica que no se suban más de 10 fotos y que cada foto no exceda los 5MB.
2. **Inserción de propiedad**: Se crea y guarda la propiedad en la base de datos.
3. **Subida de imágenes**: Se suben las imágenes a un bucket de S3 con el path especificado.
4. **Almacenamiento de fotos**: Se guarda la información de cada foto en la tabla `property_photos`.
5. **Manejo de errores**: Si ocurre un error en cualquier paso, se elimina la propiedad creada para mantener la integridad de los datos.

### Consideraciones adicionales:
- Asegúrate de tener configurado el cliente S3 con las credenciales adecuadas.
- Este código asume que tienes las entidades `Property` y `PropertyPhoto` definidas en tu proyecto.
- La gestión de errores puede ser mejorada según las necesidades específicas de tu aplicación.
